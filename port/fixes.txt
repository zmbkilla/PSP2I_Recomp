# Instruction fixes applied by allegrexrecomp emit --fixes, each a genuine bug
# in the original game. A fix must not change control flow.

# 0x08A937B4: phase accumulator step. The function advances a 16-bit phase at
# +0x98 by trunc(step(+0xC0) * scale) every frame. The original computes
# scale = fps / 30 (div.s $f1, $f1, $f0 with $f1 = fps, $f0 = 30.0), which is
# the factor for converting a DURATION in 30 Hz ticks to frames; for a
# per-frame STEP the factor is 30 / fps -- the engine's own frame-step value
# (timing state +0x18 = 30 / fps, set in 0x08B81830). At 30 fps both are 1.0,
# so the original behaviour is unchanged; at 60 fps the original runs the
# phase at 4x speed (2x per frame, 2x the frames), and at the game's own
# 20 fps mode at 0.44x. The PPSSPP 60 FPS cheat swaps the operands the same way.
0x08A937B4 0x46010043   # div.s $f1, $f0, $f1   (30 / fps)

# 0x08A5C27C: NPC walk-to-point arrival check (0x08A5BBF0 walks an NPC to a
# point: turn toward it, step forward, stop once arrived). "Arrived" is a
# horizontal distance under 30/fps -- one frame's walk at the fast speed, so
# the radius halves at 60 fps (1.0 -> 0.5). Walking (12/fps per frame) and
# turning (0x24000/fps per frame) are both fps-scaled, so the NPC's turning
# circle is 12 / 14.1 rad/s = 0.85 at any frame rate: inside the 30 fps
# radius, outside the 60 fps one, and the NPC circles the point forever
# instead of arriving -- Rapico spinning on the spot in his shop, Hyuga
# turning past the player. Logged at 60 fps: the walk restarted its turn every
# frame with the target held ~90 deg to the side (circling) for the whole
# recording. The radius keeps its 30 fps size: the division 30.0 / fps at
# 0x08A5C2B8 divides 30.0 by itself instead (fps in $f0 is not used after
# it), so 30 / 30 = 1.0 -- one instruction; the 60 fps cheat uses the same.
0x08A5C2B8 0x46010843   # div.s $f1, $f1, $f0  ->  div.s $f1, $f1, $f1   (arrival radius 30/30)

# 0x089AAD80: the lobby billboard's size limit. 0x089AAD3C hands the received
# GIM to 0x089976C4 only when its size (*0x08ED5A24) is at most 0xFFFF, so a
# bigger one leaves the billboard black. Nothing needs the limit: the HTTP
# receive (0x08A63CB8) reallocates its buffer to the reply's full length, and
# the loader (0x08997D20) allocates exactly the GIM's size and copies into
# that. The compare becomes 0, so any size is accepted (the PSP texture limits
# still apply: up to 512x512).
0x089AAD80 0x00001021   # slt $v0, $v0, $v1  ->  addu $v0, $zero, $zero# 0x08D9466C..0x08D98170: the GIM library's texture size. It rounds an
# image's width and height up to a power of two with a loop that starts at
# 512 (7 steps, 512 down to 8), so anything over 512 is declared 1024 and a
# bigger image shows only its top-left part -- the game's UVs (0x089AA7E0)
# assume the full power of two. The loops start at 2048 instead, with 9 steps
# (2048 down to 8), so up to 4096 is declared right; sizes up to 512 round as
# before. Two copies in the bind (width, height) and one helper (0x08D9815C).
0x08D9466C 0x24050800   # addiu $a1, $zero, 512  ->  2048   (width)
0x08D94688 0x28640009   # slti $a0, $v1, 7       ->  9
0x08D946B4 0x24030800   # addiu $v1, $zero, 512  ->  2048   (height)
0x08D946C4 0x28850009   # slti $a1, $a0, 7       ->  9
0x08D9815C 0x24030800   # addiu $v1, $zero, 512  ->  2048   (helper)
0x08D98170 0x28A60009   # slti $a2, $a1, 7       ->  9
